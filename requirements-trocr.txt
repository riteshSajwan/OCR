# Dependencies for the TrOCR handwriting engine (--engine trocr).
# Kept separate from requirements.txt because they are large and optional.
#
# Install torch FIRST and name the build explicitly, otherwise pip resolves to the
# ~2.5GB CUDA wheel by default:
#
#   CPU-only (~200MB):
#     .venv\Scripts\python.exe -m pip install torch --index-url https://download.pytorch.org/whl/cpu
#   CUDA:
#     .venv\Scripts\python.exe -m pip install torch --index-url https://download.pytorch.org/whl/cu124
#
# Then:
#   .venv\Scripts\python.exe -m pip install -r requirements-trocr.txt

torch
